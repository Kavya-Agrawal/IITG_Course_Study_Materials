# Create the iitg.txt file and add initial content
echo "Version 1 content" > iitg.txt

# Create an SCCS file for iitg.txt
sccs create iitg.txt

# Make and commit changes to create new versions
for i in {2..5}; do
    sccs edit iitg.txt
    echo "Version $i content" >> iitg.txt
    sccs delta iitg.txt << EOF
Comment for version $i
EOF
done

sccs cdc -r1.3 iitg.txt

sccs get -p -r1.4 iitg.txt | diff -u - iitg.txt

sccs get -p iitg.txt

sccs get -c'2024/08/28 14:30:00' iitg.txt

sccs get -x1.2 -x1.4 iitg.txt
